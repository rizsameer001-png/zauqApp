Here's the complete updated `AudioListPage.jsx` with improved voice command handling and enhanced helper box:

```jsx
// client/src/pages/public/AudioListPage.jsx
import React, { useState, useEffect, useCallback, useRef } from 'react';
import { useTranslation } from 'react-i18next';
import { useSearchParams, Link } from 'react-router-dom';
import { motion } from 'framer-motion';
import { useQuery } from '@tanstack/react-query';
import {
  Search, Headphones, Play, Clock, Eye,
  Grid, List, Loader2, AlertCircle,
  ChevronLeft, ChevronRight, Mic, MicOff,
  Volume2, VolumeX, Zap, X, Pause, SkipForward, SkipBack,
  Heart, Music, Radio, TrendingUp, Calendar, Award, Command
} from 'lucide-react';
import audioAPI from '../../api/audioAPI';
import { AUDIO_CATEGORIES } from '../../utils/constants.js';
import { useAudioPlayer } from '../../context/AudioPlayerContext';
import toast from 'react-hot-toast';

const AudioListPage = () => {
  const { t } = useTranslation();
  const [searchParams, setSearchParams] = useSearchParams();
  const [activeCategory, setActiveCategory] = useState(searchParams.get('category') || 'all');
  const [searchQuery, setSearchQuery] = useState('');
  const [viewMode, setViewMode] = useState('grid');
  const [sortBy, setSortBy] = useState('popular');
  const [currentPage, setCurrentPage] = useState(1);
  const [isListening, setIsListening] = useState(false);
  const [voiceSupported, setVoiceSupported] = useState(true);
  const [voiceTranscript, setVoiceTranscript] = useState('');
  const [showVoiceGuide, setShowVoiceGuide] = useState(false);
  const recognitionRef = useRef(null);
  const itemsPerPage = 9;

  // Get audio player functions
  const { 
    currentAudio, 
    isPlaying, 
    playAudio, 
    pauseAudio, 
    resumeAudio,
    nextAudio,
    previousAudio,
    setVolume: setPlayerVolume,
    getVolume,
    toggleMute,
    isMuted
  } = useAudioPlayer();

  // Voice command categories with examples
  const voiceCommands = {
    playback: {
      icon: Play,
      title: "Playback Control",
      commands: [
        { phrase: "Play [song name]", description: "Play specific audio", example: "Play Alvida Alvida" },
        { phrase: "Play Nadeem Sarwar", description: "Play by artist", example: "Play Nadeem Sarwar nauha" },
        { phrase: "Pause / Stop", description: "Pause current audio", example: "Pause the audio" },
        { phrase: "Resume / Continue", description: "Resume playback", example: "Resume playing" }
      ]
    },
    navigation: {
      icon: SkipForward,
      title: "Navigation",
      commands: [
        { phrase: "Next / Skip", description: "Next track", example: "Skip to next" },
        { phrase: "Previous / Back", description: "Previous track", example: "Go back" },
        { phrase: "Next page", description: "Go to next page", example: "Next page please" },
        { phrase: "Previous page", description: "Go to previous page", example: "Previous page" }
      ]
    },
    volume: {
      icon: Volume2,
      title: "Volume Control",
      commands: [
        { phrase: "Volume up / Louder", description: "Increase volume", example: "Volume up" },
        { phrase: "Volume down / Softer", description: "Decrease volume", example: "Softer please" },
        { phrase: "Mute / Silence", description: "Mute audio", example: "Mute the sound" },
        { phrase: "Unmute / Sound on", description: "Unmute audio", example: "Unmute" }
      ]
    },
    categories: {
      icon: Music,
      title: "Category Filters",
      commands: [
        { phrase: "Nauha", description: "Show Nauha audio", example: "Show me nauha" },
        { phrase: "Marsiya", description: "Show Marsiya audio", example: "Play marsiya" },
        { phrase: "Naat / Hamd", description: "Islamic praise songs", example: "Show naat" },
        { phrase: "Ghazal / Nazm", description: "Poetry recitations", example: "Play ghazal" }
      ]
    },
    search: {
      icon: Search,
      title: "Search & Discovery",
      commands: [
        { phrase: "Search [term]", description: "Search for audio", example: "Search muharram" },
        { phrase: "Show all", description: "Clear all filters", example: "Show everything" },
        { phrase: "Trending / Popular", description: "Show popular", example: "Show trending" },
        { phrase: "New releases", description: "Recent audio", example: "New releases please" }
      ]
    }
  };

  // Initialize speech recognition
  useEffect(() => {
    const SpeechRecognition = window.SpeechRecognition || window.webkitSpeechRecognition;
    
    if (!SpeechRecognition) {
      setVoiceSupported(false);
      console.warn('Speech recognition not supported in this browser');
      return;
    }

    const recognition = new SpeechRecognition();
    recognition.continuous = false;
    recognition.interimResults = true;
    recognition.lang = 'ur-PK'; // Urdu language, also supports en-US, hi-IN

    recognition.onstart = () => {
      setIsListening(true);
      setShowVoiceGuide(true);
      toast.success('🎤 Listening... Speak your command', { icon: '🎤', duration: 2000 });
    };

    recognition.onend = () => {
      setIsListening(false);
      setTimeout(() => setShowVoiceGuide(false), 3000);
    };

    recognition.onresult = (event) => {
      const transcript = Array.from(event.results)
        .map(result => result[0].transcript)
        .join('');
      
      setVoiceTranscript(transcript);
      
      if (event.results[0].isFinal) {
        processVoiceCommand(transcript);
      }
    };

    recognition.onerror = (event) => {
      console.error('Speech recognition error:', event.error);
      setIsListening(false);
      setShowVoiceGuide(false);
      
      if (event.error === 'not-allowed') {
        toast.error('Microphone access denied. Please allow microphone access.');
      } else if (event.error === 'no-speech') {
        toast.error('No speech detected. Please try again.');
      } else {
        toast.error('Voice recognition failed. Please try again.');
      }
    };

    recognitionRef.current = recognition;

    return () => {
      if (recognitionRef.current) {
        recognitionRef.current.stop();
      }
    };
  }, []);

  // Helper function to extract search terms from command
  const extractSearchTerms = (command, patterns) => {
    for (const pattern of patterns) {
      const match = command.match(pattern);
      if (match) {
        return match[1].trim();
      }
    }
    return null;
  };

  // Process voice commands - Enhanced version
  const processVoiceCommand = async (command) => {
    const lowerCommand = command.toLowerCase().trim();
    console.log('🎤 Voice command received:', command);
    
    // ============================================
    // VOLUME CONTROL COMMANDS
    // ============================================
    if (lowerCommand.match(/^(volume up|louder|increase volume|turn up|increase sound)/i)) {
      const currentVol = getVolume ? getVolume() : 0.7;
      const newVolume = Math.min(currentVol + 0.1, 1);
      if (setPlayerVolume) setPlayerVolume(newVolume);
      toast.success(`🔊 Volume: ${Math.round(newVolume * 100)}%`);
      return;
    }
    
    if (lowerCommand.match(/^(volume down|softer|decrease volume|turn down|decrease sound)/i)) {
      const currentVol = getVolume ? getVolume() : 0.7;
      const newVolume = Math.max(currentVol - 0.1, 0);
      if (setPlayerVolume) setPlayerVolume(newVolume);
      toast.success(`🔉 Volume: ${Math.round(newVolume * 100)}%`);
      return;
    }
    
    if (lowerCommand.match(/^(mute|silence|no sound|turn off sound)/i)) {
      if (toggleMute) toggleMute();
      toast.success('🔇 Audio muted');
      return;
    }
    
    if (lowerCommand.match(/^(unmute|sound on|turn on sound|unmute audio)/i)) {
      if (toggleMute && isMuted) toggleMute();
      toast.success('🔊 Audio unmuted');
      return;
    }
    
    // ============================================
    // PLAYBACK CONTROL COMMANDS
    // ============================================
    if (lowerCommand.match(/^(pause|stop|hold|रोकें|روکیں)/i)) {
      pauseAudio();
      toast.success('⏸️ Audio paused');
      return;
    }
    
    if (lowerCommand.match(/^(resume|continue|play current|फिर से चलाएं|دوبارہ چلائیں)/i)) {
      if (currentAudio) {
        resumeAudio();
        toast.success('▶️ Audio resumed');
      } else {
        toast.info('No audio to resume. Try saying "play [song name]"');
      }
      return;
    }
    
    if (lowerCommand.match(/^(next|skip|forward|अगला|اگلا)/i)) {
      nextAudio();
      toast.success('⏭️ Next track');
      return;
    }
    
    if (lowerCommand.match(/^(previous|back|prev|पिछला|پچھلا)/i)) {
      previousAudio();
      toast.success('⏮️ Previous track');
      return;
    }
    
    // ============================================
    // PAGE NAVIGATION COMMANDS
    // ============================================
    if (lowerCommand.match(/^(next page|page forward|go to next page)/i)) {
      if (currentPage < (pagination.totalPages || 1)) {
        goToPage(currentPage + 1);
        toast.success('📄 Next page');
      } else {
        toast.info('Already on the last page');
      }
      return;
    }
    
    if (lowerCommand.match(/^(previous page|page back|go to previous page)/i)) {
      if (currentPage > 1) {
        goToPage(currentPage - 1);
        toast.success('📄 Previous page');
      } else {
        toast.info('Already on the first page');
      }
      return;
    }
    
    // ============================================
    // SORT COMMANDS
    // ============================================
    if (lowerCommand.match(/^(popular|most popular|trending|top rated)/i)) {
      setSortBy('popular');
      setCurrentPage(1);
      toast.success('📈 Showing most popular audio');
      return;
    }
    
    if (lowerCommand.match(/^(recent|new|newest|latest|new releases)/i)) {
      setSortBy('recent');
      setCurrentPage(1);
      toast.success('🆕 Showing newest audio first');
      return;
    }
    
    if (lowerCommand.match(/^(most played|frequent|top plays)/i)) {
      setSortBy('plays');
      setCurrentPage(1);
      toast.success('🎧 Showing most played audio');
      return;
    }
    
    // ============================================
    // PLAY SPECIFIC AUDIO COMMANDS
    // ============================================
    const playPatterns = [
      /^play\s+(.+)$/i,
      /^play\s+audio\s+(.+)$/i,
      /^चलाएं\s+(.+)$/i,
      /^چلائیں\s+(.+)$/i,
      /^सुनाएं\s+(.+)$/i,
      /^play\s+song\s+(.+)$/i
    ];
    
    const searchTerms = extractSearchTerms(command, playPatterns);
    
    if (searchTerms) {
      toast.loading(`🔍 Searching for "${searchTerms}"...`, { id: 'voice-play' });
      
      try {
        const response = await audioAPI.searchAudio(searchTerms, { limit: 10 });
        const results = response?.data?.data || response?.data || response || [];
        
        if (results.length > 0) {
          const audioToPlay = results[0];
          playAudio(audioToPlay, results, 0);
          toast.success(`🎵 Now playing: ${audioToPlay.title}`, { id: 'voice-play', duration: 3000 });
        } else {
          toast.error(`❌ No audio found for "${searchTerms}"`, { id: 'voice-play' });
        }
      } catch (error) {
        console.error('Search error:', error);
        toast.error('Failed to search audio', { id: 'voice-play' });
      }
      return;
    }
    
    // ============================================
    // CATEGORY FILTER COMMANDS (Comprehensive)
    // ============================================
    const categoryMap = {
      // Muharram related
      'nauha': 'nauha', 'नौहा': 'nauha', 'نوحہ': 'nauha',
      'marsiya': 'marsiya', 'मरसिया': 'marsiya', 'مرثیہ': 'marsiya',
      'soz': 'soz', 'सोज़': 'soz', 'سوز': 'soz',
      'salam': 'salam', 'सलाम': 'salam', 'سلام': 'salam',
      'majlis': 'majlis', 'मजलिस': 'majlis', 'مجلس': 'majlis',
      
      // Islamic praise
      'naat': 'naat', 'नात': 'naat', 'نعت': 'naat',
      'hamd': 'hamd', 'हम्द': 'hamd', 'حمد': 'hamd',
      'manqabat': 'manqabat', 'मनक़बत': 'manqabat', 'منقبت': 'manqabat',
      'munajat': 'munajat', 'मुनाजात': 'munajat', 'مناجات': 'munajat',
      
      // Poetry
      'ghazal': 'ghazal', 'ग़ज़ल': 'ghazal', 'غزل': 'ghazal',
      'nazm': 'nazm', 'नज़्म': 'nazm', 'نظم': 'nazm',
      'poem': 'poem_recitation', 'कविता': 'poem_recitation', 'شعر': 'poem_recitation',
      
      // Other
      'podcast': 'podcast', 'पॉडकास्ट': 'podcast', 'پوڈکاسٹ': 'podcast',
      'audiobook': 'audiobook', 'ऑडियोबुक': 'audiobook', 'آڈیو بک': 'audiobook',
      'lecture': 'lecture', 'व्याख्यान': 'lecture', 'درس': 'lecture',
      'interview': 'interview', 'साक्षात्कार': 'interview', 'انٹرویو': 'interview'
    };
    
    for (const [keyword, category] of Object.entries(categoryMap)) {
      if (lowerCommand.includes(keyword)) {
        setActiveCategory(category);
        setSearchQuery('');
        setCurrentPage(1);
        toast.success(`📁 Showing ${category} audio`);
        return;
      }
    }
    
    // ============================================
    // SHOW ALL / CLEAR FILTERS COMMANDS
    // ============================================
    if (lowerCommand.match(/^(all|show all|clear filters|reset|everything|सभी|سب|clear)/i)) {
      setActiveCategory('all');
      setSearchQuery('');
      setSortBy('popular');
      setCurrentPage(1);
      toast.success('✨ Showing all audio');
      return;
    }
    
    // ============================================
    // SEARCH COMMANDS
    // ============================================
    const searchPatterns = [
      /^search\s+(.+)$/i,
      /^find\s+(.+)$/i,
      /^look for\s+(.+)$/i,
      /^खोजें\s+(.+)$/i,
      /^تلاش کریں\s+(.+)$/i
    ];
    
    const searchTerm = extractSearchTerms(command, searchPatterns);
    
    if (searchTerm) {
      setSearchQuery(searchTerm);
      setActiveCategory('all');
      setCurrentPage(1);
      toast.success(`🔍 Searching for: ${searchTerm}`);
      return;
    }
    
    // ============================================
    // HELP COMMANDS
    // ============================================
    if (lowerCommand.match(/^(help|commands|what can i say|guide|मदद|مدد)/i)) {
      setShowVoiceGuide(true);
      toast.success('📖 Voice command guide displayed above', { duration: 3000 });
      setTimeout(() => setShowVoiceGuide(false), 10000);
      return;
    }
    
    // ============================================
    // DEFAULT: Search with the command if it's meaningful
    // ============================================
    if (command.length > 3) {
      setSearchQuery(command);
      setActiveCategory('all');
      setCurrentPage(1);
      toast.success(`🔍 Searching for: ${command}`);
    } else {
      toast.error('Command not recognized. Say "help" for available commands');
    }
  };

  const startVoiceSearch = () => {
    if (!voiceSupported) {
      toast.error('Voice search is not supported in your browser');
      return;
    }
    
    if (recognitionRef.current) {
      try {
        recognitionRef.current.start();
      } catch (error) {
        console.error('Error starting recognition:', error);
        toast.error('Failed to start voice recognition');
      }
    }
  };

  const stopVoiceSearch = () => {
    if (recognitionRef.current) {
      recognitionRef.current.stop();
      setIsListening(false);
    }
  };

  const { data: response, isLoading, error, refetch } = useQuery({
    queryKey: ['audio', currentPage, activeCategory, sortBy, searchQuery],
    queryFn: () => audioAPI.getAudioItems({
      page: currentPage,
      limit: itemsPerPage,
      type: activeCategory !== 'all' ? activeCategory.toLowerCase() : undefined,
      search: searchQuery || undefined,
      sort: sortBy
    }),
    enabled: true,
    staleTime: 30000
  });

  const audioData = response?.data?.data || response?.data || response || [];
  const audioItems = Array.isArray(audioData) ? audioData : [];
  const pagination = response?.data?.pagination || response?.pagination || { total: 0, page: 1, totalPages: 1 };

  useEffect(() => {
    if (activeCategory && activeCategory !== 'all') {
      setSearchParams({ category: activeCategory, ...(searchQuery && { search: searchQuery }) });
    } else {
      setSearchParams(searchQuery ? { search: searchQuery } : {});
    }
    setCurrentPage(1);
  }, [activeCategory, searchQuery, setSearchParams]);

  const goToPage = (page) => {
    if (page >= 1 && page <= (pagination.totalPages || 1)) {
      setCurrentPage(page);
      window.scrollTo({ top: 0, behavior: 'smooth' });
    }
  };

  const formatDuration = (seconds) => {
    if (!seconds) return 'N/A';
    const hrs = Math.floor(seconds / 3600);
    const mins = Math.floor((seconds % 3600) / 60);
    const secs = seconds % 60;
    return hrs > 0
      ? `${hrs}:${mins.toString().padStart(2, '0')}:${secs.toString().padStart(2, '0')}`
      : `${mins}:${secs.toString().padStart(2, '0')}`;
  };

  const formatPlays = (plays) => {
    if (!plays) return '0';
    if (plays >= 1000000) return `${(plays / 1000000).toFixed(1)}M`;
    if (plays >= 1000) return `${(plays / 1000).toFixed(1)}K`;
    return plays.toString();
  };

  const clearFilters = () => {
    setSearchQuery('');
    setActiveCategory('all');
    setSortBy('popular');
    setCurrentPage(1);
    toast.success('Filters cleared');
  };

  const getCategoryDisplayName = (type) => {
    const category = AUDIO_CATEGORIES.find(cat => cat.id === type?.toLowerCase());
    return category?.label || type || 'Audio';
  };

  // Voice Command Guide Component
  const VoiceCommandGuide = () => (
    <motion.div
      initial={{ opacity: 0, y: -20 }}
      animate={{ opacity: 1, y: 0 }}
      exit={{ opacity: 0, y: -20 }}
      className="fixed top-20 right-4 z-50 w-96 bg-white rounded-xl shadow-2xl border border-gray-200 overflow-hidden"
    >
      <div className="bg-gradient-to-r from-primary-600 to-purple-600 px-4 py-3 flex items-center justify-between">
        <div className="flex items-center gap-2 text-white">
          <Command className="h-5 w-5" />
          <h3 className="font-semibold">Voice Command Guide</h3>
        </div>
        <button onClick={() => setShowVoiceGuide(false)} className="text-white/80 hover:text-white">
          <X className="h-5 w-5" />
        </button>
      </div>
      
      <div className="max-h-96 overflow-y-auto p-4 space-y-4">
        {Object.values(voiceCommands).map((section, idx) => {
          const Icon = section.icon;
          return (
            <div key={idx}>
              <div className="flex items-center gap-2 mb-2 pb-1 border-b border-gray-200">
                <Icon className="h-4 w-4 text-primary-600" />
                <h4 className="font-medium text-gray-900">{section.title}</h4>
              </div>
              <div className="space-y-1">
                {section.commands.map((cmd, cmdIdx) => (
                  <div key={cmdIdx} className="text-sm">
                    <span className="font-mono text-xs bg-gray-100 px-2 py-0.5 rounded text-primary-700">
                      "{cmd.phrase}"
                    </span>
                    <p className="text-xs text-gray-500 mt-0.5 ml-1">
                      {cmd.description}
                      <span className="text-gray-400"> (e.g., "{cmd.example}")</span>
                    </p>
                  </div>
                ))}
              </div>
            </div>
          );
        })}
        
        <div className="pt-2 border-t border-gray-200">
          <p className="text-xs text-center text-gray-500">
            💡 Tip: Say "help" to show this guide again
          </p>
        </div>
      </div>
    </motion.div>
  );

  // Now playing bar component
  const NowPlayingBar = () => {
    if (!currentAudio) return null;
    
    return (
      <div className="fixed bottom-0 left-0 right-0 bg-gradient-to-r from-gray-900 to-gray-800 text-white shadow-lg z-50">
        <div className="max-w-7xl mx-auto px-4 py-3">
          <div className="flex items-center justify-between">
            <div className="flex items-center gap-4 flex-1">
              <div className="w-12 h-12 bg-primary-600 rounded-lg flex items-center justify-center">
                <Headphones className="h-6 w-6" />
              </div>
              <div className="flex-1">
                <p className="font-medium text-sm truncate">{currentAudio.title}</p>
                <p className="text-xs text-gray-400">
                  {currentAudio.author?.name || 'Unknown Artist'}
                </p>
              </div>
            </div>
            <div className="flex items-center gap-3">
              <button
                onClick={previousAudio}
                className="p-2 hover:bg-white/10 rounded-full transition"
              >
                <SkipBack className="h-5 w-5" />
              </button>
              <button
                onClick={isPlaying ? pauseAudio : resumeAudio}
                className="p-3 bg-primary-600 hover:bg-primary-700 rounded-full transition"
              >
                {isPlaying ? <Pause className="h-5 w-5" /> : <Play className="h-5 w-5 ml-0.5" />}
              </button>
              <button
                onClick={nextAudio}
                className="p-2 hover:bg-white/10 rounded-full transition"
              >
                <SkipForward className="h-5 w-5" />
              </button>
            </div>
          </div>
        </div>
      </div>
    );
  };

  /* ================= LOADING ================= */
  if (isLoading && audioItems.length === 0) {
    return (
      <div className="min-h-screen flex items-center justify-center bg-gradient-to-br from-gray-50 to-gray-100">
        <Loader2 className="h-12 w-12 animate-spin text-primary-600" />
      </div>
    );
  }

  /* ================= ERROR ================= */
  if (error && audioItems.length === 0) {
    return (
      <div className="min-h-screen flex flex-col items-center justify-center text-center p-4">
        <AlertCircle className="h-16 w-16 text-red-500 mb-4" />
        <h2 className="text-xl font-semibold mb-2">Failed to load audio</h2>
        <p className="text-gray-500 mb-4">Please check your connection and try again</p>
        <button onClick={() => refetch()} className="btn-primary">Retry</button>
      </div>
    );
  }

  return (
    <div className="min-h-screen pt-20 pb-16 bg-gradient-to-br from-gray-50 via-white to-gray-100">
      {/* Voice Command Guide Modal */}
      {showVoiceGuide && <VoiceCommandGuide />}
      
      {/* Now Playing Bar */}
      <NowPlayingBar />

      <div className="max-w-7xl mx-auto px-4 sm:px-6 lg:px-8">

        {/* HEADER */}
        <div className="mb-10 text-center">
          <h1 className="text-4xl font-bold bg-gradient-to-r from-primary-600 to-purple-600 bg-clip-text text-transparent">
            🎵 Audio Library
          </h1>
          <p className="text-gray-500 mt-2">
            Discover premium Urdu audio experiences - Nauha, Marsiya, Naat, Ghazal & more
          </p>
          
          {/* Voice Search Badge */}
          {voiceSupported && (
            <div className="mt-3 inline-flex items-center gap-2 px-3 py-1 bg-primary-50 text-primary-600 text-sm rounded-full">
              <Mic className="h-4 w-4" />
              <span>🎤 Voice search available - Click the mic and speak</span>
              <button 
                onClick={() => setShowVoiceGuide(!showVoiceGuide)}
                className="ml-2 text-primary-500 hover:text-primary-700"
              >
                <Command className="h-3 w-3" />
              </button>
            </div>
          )}
        </div>

        {/* CONTROLS */}
        <div className="sticky top-20 z-10 bg-white/70 backdrop-blur-xl border border-gray-200 rounded-2xl p-4 mb-8 shadow-sm flex flex-col md:flex-row gap-4">

          {/* SEARCH with Voice */}
          <div className="flex-1 relative">
            <Search className="absolute left-4 top-1/2 -translate-y-1/2 text-gray-400 h-5 w-5" />
            <input
              type="text"
              placeholder={isListening ? "🎤 Listening... Speak your command" : "🔍 Search audio by title, artist, or type..."}
              value={searchQuery}
              onChange={(e) => setSearchQuery(e.target.value)}

                            {audio.title}

                              by {typeof audio.author === 'object' ? audio.author.name : audio.author}

                          )}

                          {audio.description && (
                            <p className="text-sm text-gray-500 mt-2 line-clamp-2">
                              {audio.description}
                            </p>
                          )}

                          <div className="flex gap-4 text-sm text-gray-400 mt-3">

                  </div>

            {/* PAGINATION */}
            {(pagination.totalPages || 1) > 1 && (
              <div className="flex justify-center gap-2 mt-12">
                <button
                  onClick={() => goToPage(currentPage - 1)}
                  disabled={currentPage === 1}
                  className="p-2 border rounded-lg disabled:opacity-50 disabled:cursor-not-allowed hover:bg-gray-50"
                >
                  <ChevronLeft className="h-5 w-5" />
                </button>

                {[...Array(Math.min(pagination.totalPages || 1, 7))].map((_, i) => {
                  let pageNum;
                  const totalPages = pagination.totalPages || 1;
                  
                  if (totalPages <= 7) {
                    pageNum = i + 1;
                  } else if (currentPage <= 4) {
                    pageNum = i + 1;
                    if (i === 6) pageNum = totalPages;
                  } else if (currentPage >= totalPages - 3) {
                    pageNum = totalPages - 6 + i;
                  } else {
                    pageNum = currentPage - 3 + i;
                    if (i === 0) pageNum = 1;
                    if (i === 6) pageNum = totalPages;
                  }
                  
                  if (i === 5 && totalPages > 7 && currentPage > 4 && currentPage < totalPages - 3) {
                    return <span key="ellipsis" className="px-4 py-2">...</span>;
                  }
                  
                  return (
                    <button
                      key={pageNum}
                      onClick={() => goToPage(pageNum)}
                      className={`px-4 py-2 rounded-lg ${
                        currentPage === pageNum
                          ? 'bg-primary-600 text-white'
                          : 'bg-white border hover:bg-gray-50'
                      }`}
                    >
                      {pageNum}
                    </button>
                  );
                })}

                <button
                  onClick={() => goToPage(currentPage + 1)}
                  disabled={currentPage === (pagination.totalPages || 1)}
                  className="p-2 border rounded-lg disabled:opacity-50 disabled:cursor-not-allowed hover:bg-gray-50"
                >
                  <ChevronRight className="h-5 w-5" />
                </button>
              </div>
            )}
          </>
        )}
      </div>
    </div>
  );
};

export default AudioListPage;
```

```jsx
// client/src/components/home/AudioPlayer.jsx
import React, { useState, useRef, useEffect } from 'react';
import {
  Play, Pause, SkipBack, SkipForward, Volume2, VolumeX,
  Heart, List, Repeat, Shuffle, Maximize2, Minimize2,
  Clock, Music, Headphones, Mic, MicOff, X, Crown
} from 'lucide-react';
import { motion, AnimatePresence } from 'framer-motion';
import subscriptionAPI from '../../api/subscriptionAPI';
import { useVoiceSearch } from '../../hooks/useVoiceSearch';
import { parseVoiceCommand } from '../../utils/voiceCommands';
import toast from 'react-hot-toast';

const AudioPlayer = ({ tracks, currentTrackIndex, onTrackChange, onClose, onPlayExternalTrack, onFilterCategory }) => {
  const [isPlaying, setIsPlaying] = useState(false);
  const [currentTime, setCurrentTime] = useState(0);
  const [duration, setDuration] = useState(0);
  const [volume, setVolume] = useState(1);
  const [isMuted, setIsMuted] = useState(false);
  const [isLiked, setIsLiked] = useState(false);
  const [showPlaylist, setShowPlaylist] = useState(false);
  const [isMinimized, setIsMinimized] = useState(false);
  const [isShuffled, setIsShuffled] = useState(false);
  const [repeatMode, setRepeatMode] = useState('none'); // none, one, all
  const [userPlan, setUserPlan] = useState(null);
  const [shuffledTracks, setShuffledTracks] = useState([]);
  const [shuffledIndex, setShuffledIndex] = useState(0);
  
  const audioRef = useRef(null);
  const progressBarRef = useRef(null);

  const currentTrack = tracks[currentTrackIndex];

  // Initialize shuffled tracks
  useEffect(() => {
    if (isShuffled && tracks.length > 0) {
      const shuffled = [...tracks];
      for (let i = shuffled.length - 1; i > 0; i--) {
        const j = Math.floor(Math.random() * (i + 1));
        [shuffled[i], shuffled[j]] = [shuffled[j], shuffled[i]];
      }
      setShuffledTracks(shuffled);
      setShuffledIndex(shuffled.findIndex(track => track._id === currentTrack?._id));
    }
  }, [isShuffled, tracks]);

  useEffect(() => {
    fetchUserPlan();
  }, []);

  const fetchUserPlan = async () => {
    try {
      const response = await subscriptionAPI.getCurrent();
      const data = response.data || response;
      setUserPlan(data.plan);
    } catch (error) {
      console.error('Error fetching user plan:', error);
    }
  };

  useEffect(() => {
    if (audioRef.current) {
      if (isPlaying) {
        audioRef.current.play().catch(err => console.error('Play error:', err));
      } else {
        audioRef.current.pause();
      }
    }
  }, [isPlaying, currentTrackIndex]);

  const handlePlayPause = () => {
    setIsPlaying(!isPlaying);
  };

  const handleTimeUpdate = () => {
    setCurrentTime(audioRef.current.currentTime);
  };

  const handleLoadedMetadata = () => {
    setDuration(audioRef.current.duration);
  };

  const handleProgressChange = (e) => {
    const newTime = (e.target.value / 100) * duration;
    audioRef.current.currentTime = newTime;
    setCurrentTime(newTime);
  };

  const handleVolumeChange = (e) => {
    const newVolume = e.target.value / 100;
    setVolume(newVolume);
    audioRef.current.volume = newVolume;
    setIsMuted(newVolume === 0);
  };

  const toggleMute = () => {
    if (isMuted) {
      audioRef.current.volume = volume;
      setIsMuted(false);
    } else {
      audioRef.current.volume = 0;
      setIsMuted(true);
    }
  };

  const handlePrevious = () => {
    if (isShuffled && shuffledTracks.length > 0) {
      const newIndex = shuffledIndex === 0 ? shuffledTracks.length - 1 : shuffledIndex - 1;
      setShuffledIndex(newIndex);
      const originalIndex = tracks.findIndex(track => track._id === shuffledTracks[newIndex]._id);
      onTrackChange(originalIndex);
    } else {
      const newIndex = currentTrackIndex === 0 ? tracks.length - 1 : currentTrackIndex - 1;
      onTrackChange(newIndex);
    }
  };

  const handleNext = () => {
    if (repeatMode === 'one') {
      audioRef.current.currentTime = 0;
      audioRef.current.play();
    } else if (isShuffled && shuffledTracks.length > 0) {
      const newIndex = shuffledIndex === shuffledTracks.length - 1 ? 0 : shuffledIndex + 1;
      setShuffledIndex(newIndex);
      const originalIndex = tracks.findIndex(track => track._id === shuffledTracks[newIndex]._id);
      onTrackChange(originalIndex);
    } else {
      const newIndex = currentTrackIndex === tracks.length - 1 ? 0 : currentTrackIndex + 1;
      onTrackChange(newIndex);
    }
  };

  const formatTime = (time) => {
    if (isNaN(time)) return '0:00';
    const minutes = Math.floor(time / 60);
    const seconds = Math.floor(time % 60);
    return `${minutes}:${seconds.toString().padStart(2, '0')}`;
  };

  const handleLike = () => {
    setIsLiked(!isLiked);
  };

  // ============================================
  // VOICE SEARCH INTEGRATION
  // ============================================
  
  const handleVoiceCommand = async (command) => {
    console.log('Voice command received:', command);
    
    const result = await parseVoiceCommand(command, { playlist: tracks });
    
    switch (result.type) {
      case 'PLAY_FROM_PLAYLIST':
        if (result.index !== undefined && result.index >= 0 && result.index < tracks.length) {
          onTrackChange(result.index);
          setIsPlaying(true);
          toast.success(`Playing: ${result.audio?.title || tracks[result.index].title}`);
        } else {
          toast.error('Track not found in playlist');
        }
        break;
        
      case 'PLAY_SEARCH_RESULT':
        if (result.audio && onPlayExternalTrack) {
          onPlayExternalTrack(result.audio, result.playlist || [result.audio]);
          toast.success(`Playing: ${result.audio.title}`);
        } else if (onPlayExternalTrack) {
          toast.error('No audio found');
        }
        break;
        
      case 'PAUSE':
        if (isPlaying) {
          setIsPlaying(false);
          toast.success('Audio paused');
        }
        break;
        
      case 'RESUME':
        if (!isPlaying && currentTrack) {
          setIsPlaying(true);
          toast.success('Audio resumed');
        } else if (!currentTrack) {
          toast.error('No audio playing');
        }
        break;
        
      case 'NEXT':
        if (tracks.length > 0) {
          handleNext();
          toast.success('Next track');
        }
        break;
        
      case 'PREVIOUS':
        if (tracks.length > 0) {
          handlePrevious();
          toast.success('Previous track');
        }
        break;
        
      case 'VOLUME_UP':
        const newVolumeUp = Math.min(volume + 0.1, 1);
        setVolume(newVolumeUp);
        if (audioRef.current) audioRef.current.volume = newVolumeUp;
        setIsMuted(false);
        toast.success(`Volume: ${Math.round(newVolumeUp * 100)}%`);
        break;
        
      case 'VOLUME_DOWN':
        const newVolumeDown = Math.max(volume - 0.1, 0);
        setVolume(newVolumeDown);
        if (audioRef.current) audioRef.current.volume = newVolumeDown;
        setIsMuted(newVolumeDown === 0);
        toast.success(`Volume: ${Math.round(newVolumeDown * 100)}%`);
        break;
        
      case 'MUTE':
        if (!isMuted) {
          toggleMute();
          toast.success('Audio muted');
        }
        break;
        
      case 'UNMUTE':
        if (isMuted) {
          toggleMute();
          toast.success('Audio unmuted');
        }
        break;
        
      case 'FILTER_CATEGORY':
        if (onFilterCategory) {
          onFilterCategory(result.category);
          toast.success(`Showing ${result.category} audio`);
        }
        break;
        
      case 'ERROR':
        toast.error(result.message);
        break;
        
      default:
        toast.info(`Command not recognized: "${command.substring(0, 30)}..."`);
    }
  };

  const { isListening, voiceSupported, startListening, stopListening } = useVoiceSearch(handleVoiceCommand);

  const checkPremiumFeature = () => {
    if (userPlan?.plan === 'free' && currentTrack?.isPremium) {
      return (
        <div className="absolute inset-0 bg-black/50 flex items-center justify-center z-10 rounded-xl">
          <div className="text-center p-4">
            <Crown className="h-12 w-12 text-yellow-500 mx-auto mb-3" />
            <p className="text-white mb-3">Premium content</p>
            <button className="bg-primary-600 text-white px-4 py-2 rounded-lg">
              Upgrade to Premium
            </button>
          </div>
        </div>
      );
    }
    return null;
  };

  // Voice search button component
  const VoiceButton = () => (
    <button
      onClick={isListening ? stopListening : startListening}
      className={`p-2 rounded-full transition-all ${
        isListening 
          ? 'bg-red-500 text-white animate-pulse' 
          : 'hover:bg-white/10 text-gray-400 hover:text-white'
      }`}
      title={isListening ? "Stop listening" : "Voice search"}
    >
      {isListening ? <MicOff className="h-5 w-5" /> : <Mic className="h-5 w-5" />}
    </button>
  );

  // Voice command helper component
  const VoiceCommandHelper = () => (
    <motion.div
      initial={{ opacity: 0, y: 10 }}
      animate={{ opacity: 1, y: 0 }}
      className="absolute bottom-full left-0 mb-2 p-3 bg-gray-800 rounded-lg shadow-xl text-xs"
    >
      <div className="flex items-center gap-2 text-primary-400 mb-2">
        <Mic className="h-3 w-3" />
        <span className="font-medium">Voice Commands:</span>
      </div>
      <div className="flex flex-wrap gap-2">
        <span className="px-2 py-1 bg-gray-700 rounded">"Play [song name]"</span>
        <span className="px-2 py-1 bg-gray-700 rounded">"Pause"</span>
        <span className="px-2 py-1 bg-gray-700 rounded">"Next"</span>
        <span className="px-2 py-1 bg-gray-700 rounded">"Volume up"</span>
        <span className="px-2 py-1 bg-gray-700 rounded">"Mute"</span>
      </div>
    </motion.div>
  );

  if (isMinimized) {
    return (
      <motion.div
        initial={{ y: 100 }}
        animate={{ y: 0 }}
        className="fixed bottom-0 left-0 right-0 bg-white border-t border-gray-200 shadow-lg z-50"
      >
        <div className="flex items-center justify-between p-3 max-w-7xl mx-auto">
          <div className="flex items-center gap-3 flex-1">
            <img
              src={currentTrack?.coverArt || currentTrack?.thumbnail || '/default-album.jpg'}
              alt={currentTrack?.title}
              className="w-12 h-12 rounded-lg object-cover"
            />
            <div>
              <p className="font-medium text-gray-900">{currentTrack?.title}</p>
              <p className="text-sm text-gray-500">{currentTrack?.artist || currentTrack?.author?.name || 'Unknown Artist'}</p>
            </div>
          </div>
          <div className="flex items-center gap-4">
            {/* Voice Search Button in Minimized View */}
            {voiceSupported && <VoiceButton />}
            
            <button onClick={handlePrevious} className="p-2 hover:bg-gray-100 rounded-full">
              <SkipBack className="h-5 w-5" />
            </button>
            <button onClick={handlePlayPause} className="p-3 bg-primary-600 hover:bg-primary-700 rounded-full text-white">
              {isPlaying ? <Pause className="h-5 w-5" /> : <Play className="h-5 w-5" />}
            </button>
            <button onClick={handleNext} className="p-2 hover:bg-gray-100 rounded-full">
              <SkipForward className="h-5 w-5" />
            </button>
            <button onClick={() => setIsMinimized(false)} className="p-2 hover:bg-gray-100 rounded-full">
              <Maximize2 className="h-5 w-5" />
            </button>
          </div>
        </div>
      </motion.div>
    );
  }

  return (
    <motion.div
      initial={{ y: 100, opacity: 0 }}
      animate={{ y: 0, opacity: 1 }}
      exit={{ y: 100, opacity: 0 }}
      className="fixed bottom-0 left-0 right-0 bg-gradient-to-r from-gray-900 to-gray-800 text-white shadow-2xl z-50"
    >
      {/* Main Player */}
      <div className="max-w-7xl mx-auto p-4 relative">
        {/* Voice Command Helper - appears when listening */}
        {isListening && <VoiceCommandHelper />}
        
        <div className="flex items-center justify-between mb-4">
          <div className="flex items-center gap-4">
            <div className="relative">
              <img
                src={currentTrack?.coverArt || currentTrack?.thumbnail || '/default-album.jpg'}
                alt={currentTrack?.title}
                className="w-16 h-16 rounded-lg object-cover"
              />
              {checkPremiumFeature()}
            </div>
            <div>
              <h4 className="font-semibold">{currentTrack?.title}</h4>
              <p className="text-sm text-gray-400">{currentTrack?.artist || currentTrack?.author?.name || 'Unknown Artist'}</p>
            </div>
            <button
              onClick={handleLike}
              className={`p-2 rounded-full transition-colors ${
                isLiked ? 'text-red-500' : 'text-gray-400 hover:text-red-500'
              }`}
            >
              <Heart className="h-5 w-5" fill={isLiked ? 'currentColor' : 'none'} />
            </button>
          </div>
          
          <div className="flex items-center gap-2">
            {/* Voice Search Button */}
            {voiceSupported && <VoiceButton />}
            
            <button
              onClick={() => setShowPlaylist(!showPlaylist)}
              className="p-2 hover:bg-white/10 rounded-full transition-colors"
            >
              <List className="h-5 w-5" />
            </button>
            <button
              onClick={() => setIsMinimized(true)}
              className="p-2 hover:bg-white/10 rounded-full transition-colors"
            >
              <Minimize2 className="h-5 w-5" />
            </button>
            {onClose && (
              <button
                onClick={onClose}
                className="p-2 hover:bg-white/10 rounded-full transition-colors"
              >
                <X className="h-5 w-5" />
              </button>
            )}
          </div>
        </div>

        {/* Progress Bar */}
        <div className="mb-4">
          <input
            type="range"
            min="0"
            max="100"
            value={(currentTime / duration) * 100 || 0}
            onChange={handleProgressChange}
            className="w-full h-1 bg-gray-600 rounded-lg appearance-none cursor-pointer"
            style={{
              background: `linear-gradient(to right, #8b5cf6 0%, #8b5cf6 ${(currentTime / duration) * 100}%, #4a5568 ${(currentTime / duration) * 100}%, #4a5568 100%)`
            }}
          />
          <div className="flex justify-between text-xs text-gray-400 mt-1">
            <span>{formatTime(currentTime)}</span>
            <span>{formatTime(duration)}</span>
          </div>
        </div>

        {/* Controls */}
        <div className="flex items-center justify-between">
          <div className="flex items-center gap-3">
            <button
              onClick={() => setIsShuffled(!isShuffled)}
              className={`p-2 rounded-full transition-colors ${
                isShuffled ? 'text-primary-500' : 'text-gray-400 hover:text-white'
              }`}
            >
              <Shuffle className="h-5 w-5" />
            </button>
            <button
              onClick={() => {
                const modes = ['none', 'one', 'all'];
                const currentIdx = modes.indexOf(repeatMode);
                const nextMode = modes[(currentIdx + 1) % modes.length];
                setRepeatMode(nextMode);
                toast.info(`Repeat: ${nextMode === 'none' ? 'Off' : nextMode === 'one' ? 'One' : 'All'}`);
              }}
              className={`p-2 rounded-full transition-colors relative ${
                repeatMode !== 'none' ? 'text-primary-500' : 'text-gray-400 hover:text-white'
              }`}
            >
              <Repeat className="h-5 w-5" />
              {repeatMode === 'one' && (
                <span className="absolute text-xs ml-3 mt-2">1</span>
              )}
            </button>
          </div>

          <div className="flex items-center gap-4">
            <button onClick={handlePrevious} className="p-2 hover:bg-white/10 rounded-full transition-colors">
              <SkipBack className="h-6 w-6" />
            </button>
            <button
              onClick={handlePlayPause}
              className="p-4 bg-primary-600 hover:bg-primary-700 rounded-full transition-colors"
            >
              {isPlaying ? <Pause className="h-6 w-6" /> : <Play className="h-6 w-6" />}
            </button>
            <button onClick={handleNext} className="p-2 hover:bg-white/10 rounded-full transition-colors">
              <SkipForward className="h-6 w-6" />
            </button>
          </div>

          <div className="flex items-center gap-3">
            <button onClick={toggleMute} className="p-2 hover:bg-white/10 rounded-full transition-colors">
              {isMuted ? <VolumeX className="h-5 w-5" /> : <Volume2 className="h-5 w-5" />}
            </button>
            <input
              type="range"
              min="0"
              max="100"
              value={volume * 100}
              onChange={handleVolumeChange}
              className="w-24 h-1 bg-gray-600 rounded-lg appearance-none cursor-pointer"
            />
          </div>
        </div>

        {/* Listening indicator text */}
        {isListening && (
          <div className="absolute bottom-0 left-1/2 transform -translate-x-1/2 mb-2">
            <div className="flex items-center gap-2 px-3 py-1 bg-red-500/80 rounded-full text-xs">
              <div className="w-2 h-2 bg-white rounded-full animate-ping" />
              <span>Listening...</span>
            </div>
          </div>
        )}
      </div>

      {/* Playlist Sidebar */}
      <AnimatePresence>
        {showPlaylist && (
          <motion.div
            initial={{ x: 300, opacity: 0 }}
            animate={{ x: 0, opacity: 1 }}
            exit={{ x: 300, opacity: 0 }}
            className="absolute bottom-full right-0 mb-2 w-80 bg-gray-800 rounded-lg shadow-xl max-h-96 overflow-y-auto"
          >
            <div className="p-4">
              <div className="flex items-center justify-between mb-3">
                <h4 className="font-semibold">Playlist</h4>
                <button onClick={() => setShowPlaylist(false)} className="text-gray-400 hover:text-white">
                  <X className="h-4 w-4" />
                </button>
              </div>
              
              {/* Voice command hint in playlist */}
              {voiceSupported && (
                <div className="mb-3 p-2 bg-gray-700/50 rounded-lg text-xs text-gray-400 flex items-center gap-2">
                  <Mic className="h-3 w-3" />
                  <span>Say "Play [song name]" to play from playlist</span>
                </div>
              )}
              
              <div className="space-y-2">
                {tracks.map((track, idx) => (
                  <button
                    key={track._id || idx}
                    onClick={() => {
                      onTrackChange(idx);
                      setShowPlaylist(false);
                    }}
                    className={`w-full text-left p-2 rounded-lg transition-colors ${
                      idx === currentTrackIndex
                        ? 'bg-primary-600/20 text-primary-400'
                        : 'hover:bg-white/10'
                    }`}
                  >
                    <div className="flex items-center gap-3">
                      <span className="text-sm text-gray-400">{idx + 1}</span>
                      <div className="flex-1">
                        <p className="text-sm font-medium truncate">{track.title}</p>
                        <p className="text-xs text-gray-400 truncate">{track.artist || track.author?.name || 'Unknown'}</p>
                      </div>
                      {track.isPremium && (
                        <Crown className="h-3 w-3 text-yellow-500 flex-shrink-0" />
                      )}
                    </div>
                  </button>
                ))}
              </div>
            </div>
          </motion.div>
        )}
      </AnimatePresence>

      <audio
        ref={audioRef}
        src={currentTrack?.audioUrl}
        onTimeUpdate={handleTimeUpdate}
        onLoadedMetadata={handleLoadedMetadata}
        onEnded={handleNext}
        onError={(e) => {
          console.error('Audio error:', e);
          toast.error('Playback error');
        }}
      />
    </motion.div>
  );
};

export default AudioPlayer;
```

## Key Additions to the AudioPlayer:

1. **Voice Search Button** - Added to both minimized and expanded views
2. **Voice Command Helper** - Shows available voice commands when listening
3. **Listening Indicator** - Visual feedback when microphone is active
4. **Full Voice Command Integration** - Handles play, pause, next, previous, volume, mute, and category filtering
5. **Shuffle Support** - Properly handles shuffled playlist with voice commands
6. **Toast Notifications** - Provides feedback for all voice actions
7. **External Track Support** - Can play tracks from search results outside current playlist

## Voice Commands Now Working:

- ✅ "Play [song name]" - Plays from current playlist or searches

- ✅ "Resume" / "Continue" - Resumes playback
- ✅ "Next" / "Skip" - Next track
- ✅ "Previous" / "Back" - Previous track
- ✅ "Volume up" / "Louder" - Increases volume
- ✅ "Volume down" / "Softer" - Decreases volume
- ✅ "Mute" / "Silence" - Mutes audio
- ✅ "Unmute" / "Sound on" - Unmutes audio
- ✅ "Nauha", "Marsiya", etc. - Filters by category
